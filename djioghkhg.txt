<script>
const codeMap = {
  0:"Açıq səma", 1:"Əsasən açıq", 2:"Qismən buludlu", 3:"Tutqun",
  45:"Duman", 48:"Şaxtalı duman",
  51:"Yüngül çiskin", 53:"Orta çiskin", 55:"Sıx çiskin",
  61:"Yüngül yağış", 63:"Orta yağış", 65:"Güclü yağış",
  71:"Yüngül qar", 73:"Orta qar", 75:"Güclü qar",
  80:"Yüngül leysan", 81:"Orta leysan", 82:"Güclü leysan",
  95:"Tufan", 96:"Dolu ilə tufan", 99:"Güclü dolu ilə tufan"
};

const form = document.getElementById('searchForm');
const input = document.getElementById('cityInput');
const result = document.getElementById('result');

form.addEventListener('submit', async (e) => {
  e.preventDefault();
  const city = input.value.trim();
  if(!city) return;

  result.innerHTML = '<p class="msg loading">Yüklənir...</p>';

  try {
    // 1) Şəhərin koordinatlarını tapmaq (geokodlaşdırma)
    const geoRes = await fetch(
      `https://geocoding-api.open-meteo.com/v1/search?name=${encodeURIComponent(city)}&count=1&language=az&format=json`
    );
    const geoData = await geoRes.json();

    if (!geoData.results || geoData.results.length === 0) {
      result.innerHTML = '<p class="msg">Şəhər tapılmadı. Başqa ad yoxlayın.</p>';
      return;
    }

    const { latitude, longitude, name, country } = geoData.results[0];

    // 2) Həmin koordinatlar üçün cari hava məlumatı
    const weatherRes = await fetch(
      `https://api.open-meteo.com/v1/forecast?latitude=${latitude}&longitude=${longitude}&current=temperature_2m,relative_humidity_2m,wind_speed_10m,weather_code&timezone=auto`
    );
    const weatherData = await weatherRes.json();
    const cur = weatherData.current;

    const desc = codeMap[cur.weather_code] || "Naməlum";

    result.innerHTML = `
      <div class="desc">${name}, ${country}</div>
      <div class="temp">${Math.round(cur.temperature_2m)}°C</div>
      <div class="desc">${desc}</div>
      <div class="grid">
        <div class="stat">Nəm<b>${cur.relative_humidity_2m}%</b></div>
        <div class="stat">Külək<b>${cur.wind_speed_10m} km/s</b></div>
      </div>
    `;
  } catch (err) {
    result.innerHTML = '<p class="msg">Xəta baş verdi. İnternet bağlantısını yoxlayın.</p>';
  }
});
</script>
